\section{Complete Practical Coding}
\label{sec:practical-01}

This complete report is a reusable first project.

\begin{latexcode}{Complete practical file: first short report}
\documentclass[11pt]{article}

\title{A Short Observation Report}
\author{Your Name}
\date{\today}

\begin{document}
\maketitle

\begin{abstract}
This report records one objective, one method, and one result.
\end{abstract}

\section{Objective}
Compare two sets of demonstration measurements.

\section{Method}
Record each value under the same conditions.

\section{Result}
Group B had the larger demonstration mean.

\section{Conclusion}
The source compiled and the report has a clear structure.

\end{document}
\end{latexcode}

\begin{codelab}

\end{codelab}
